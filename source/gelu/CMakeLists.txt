if(USE_OPENCL_CPP)
    add_executable(gelu main.cpp opencl_cpp/gelu.cpp)
    target_link_libraries(gelu PRIVATE opencl_cpp_helpers)
    target_compile_definitions(gelu PRIVATE GPU_OPENCL_CPP_BACKEND)
elseif(USE_OPENCL)
    add_executable(gelu main.cpp opencl/gelu.cpp)
    target_link_libraries(gelu PRIVATE opencl_helpers)
    target_compile_definitions(gelu PRIVATE GPU_OPENCL_BACKEND)
else()
    set_source_files_properties(cuda/gelu.cpp main.cpp PROPERTIES LANGUAGE CUDA)
    add_executable(gelu main.cpp cuda/gelu.cpp)
    target_link_libraries(gelu PRIVATE cuda_helpers)
endif()
